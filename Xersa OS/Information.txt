5. Create the Networks
# NAT Network with its own DHCP
& $vbm natnetwork add --netname labnet --network 10.10.10.0/24 --enable --dhcp on

# Host-only network (VirtualBox 7.x)
& $vbm hostonlynet add --name labhost --netmask 255.255.255.0 --lower-ip 192.168.56.100 --upper-ip 192.168.56.200 --enable

# The internal network needs no creation: any VM naming it joins it.

& $vbm modifyvm $n --memory 2048 --cpus 2 --vram 16 --firmware efi `
  --nic1 natnetwork --nat-network1 labnet `
  --nic2 hostonly --host-only-adapter2 $hostif `
  --audio-driver none --clipboard-mode bidirectional
& $vbm createmedium disk --filename "$lab/vms/$n/$n.vdi" --size 20480
& $vbm storagectl $n --name SATA --add sata --controller IntelAhci
& $vbm storageattach $n --storagectl SATA --port 0 --device 0 --type hdd --medium "$lab/vms/$n/$n.vdi"
& $vbm storageattach $n --storagectl SATA --port 1 --device 0 --type dvddrive --medium $iso
& $vbm showvminfo $n --machinereadable | Select-String 'name=|memory=|cpus=|nic1=|nic2='
